# Source: tools/make-textures.py

**Packs the tile sprites into the atlas; draws the node card textures.** Explained in [world](../world.md). [On GitHub](https://github.com/LuckyMonkey/luckymonkey/blob/main/tools/make-textures.py) (needs repo access). Generated by `tools/wiki-source.py`; back to the [source index](index.md).

```python
#!/usr/bin/env python3
"""Bake the Bevy game's textures into game/assets/ (run from the repo root).

tiles.png  - 4x4 atlas of the 32px tile sprites, each cell padded 1px by edge
             extension so diamond UVs never bleed; cell 15 is plain white (walls).
card.png / card_border.png - node card fill + outline at 2x (160px, radius 16).
"""
from PIL import Image, ImageDraw

TILES = ["grass1", "grass2", "grass3", "grass4",
         "water", "water1", "water3", "water4", "water5",
         "lava1", "lava2", "lava3", "lava4", "lava_tile", "temple_tile"]
CELL = 34
atlas = Image.new("RGBA", (CELL * 4, CELL * 4), (0, 0, 0, 0))
for i, name in enumerate(TILES + ["white"]):
    src = Image.new("RGBA", (32, 32), (255, 255, 255, 255)) if name == "white" else \
        Image.open(f"public/sprites/{name}.png").convert("RGBA")
    padded = src.resize((34, 34), Image.NEAREST)  # placeholder size
    padded.paste(src.crop((0, 0, 32, 1)).resize((32, 1)), (1, 0))
    padded.paste(src.crop((0, 31, 32, 32)), (1, 33))
    padded.paste(src.crop((0, 0, 1, 32)), (0, 1))
    padded.paste(src.crop((31, 0, 32, 32)), (33, 1))
    for (cx, cy), (sx, sy) in {(0, 0): (0, 0), (33, 0): (31, 0), (0, 33): (0, 31), (33, 33): (31, 31)}.items():
        padded.putpixel((cx, cy), src.getpixel((sx, sy)))
    padded.paste(src, (1, 1))
    atlas.paste(padded, ((i % 4) * CELL, (i // 4) * CELL))
atlas.save("game/assets/tiles.png")

S = 160
fill = Image.new("RGBA", (S, S), (0, 0, 0, 0))
ImageDraw.Draw(fill).rounded_rectangle((0, 0, S - 1, S - 1), radius=16, fill=(255, 255, 255, 255))
fill.save("game/assets/card.png")
border = Image.new("RGBA", (S, S), (0, 0, 0, 0))
ImageDraw.Draw(border).rounded_rectangle((1, 1, S - 2, S - 2), radius=16, outline=(255, 255, 255, 255), width=4)
border.save("game/assets/card_border.png")

print("textures ok")
```
